2. 工作流引擎
typescript
// src/core/engine.ts
import { Skill, SkillResult } from '../types';
import { SkillRegistry } from './registry';
import { Logger } from '../utils/logger';
import { ConfigManager } from '../config/manager';

export class WorkflowEngine {
  private skillRegistry: SkillRegistry;
  private logger: Logger;
  private configManager: ConfigManager;

  constructor() {
    this.skillRegistry = SkillRegistry.getInstance();
    this.logger = new Logger();
    this.configManager = new ConfigManager();
  }

  async executeSkill(skillId: string, options: any): Promise<SkillResult> {
    const skill = this.skillRegistry.getSkill(skillId);
    
    if (!skill) {
      throw new Error(`技能 ${skillId} 未找到`);
    }

    this.logger.start(`执行技能: ${skill.name}`);
    
    try {
      const context = await this.buildContext(options);
      const result = await skill.execute(context);
      
      this.logger.success(`技能执行完成`);
      await this.postProcess(result);
      
      return result;
    } catch (error) {
      this.logger.error(`技能执行失败: ${error.message}`);
      throw error;
    }
  }

  private async buildContext(options: any) {
    const config = await this.configManager.getConfig();
    
    return {
      options,
      config,
      workspace: process.cwd(),
      timestamp: new Date().toISOString(),
      env: {
        nodeEnv: process.env.NODE_ENV,
        // 添加其他环境变量
      }
    };
  }

  private async postProcess(result: SkillResult) {
    // 保存执行历史
    await this.configManager.saveHistory({
      skillId: result.skillId,
      timestamp: new Date().toISOString(),
      success: result.success,
      output: result.output
    });

    // 发送通知（可选）
    if (result.notify) {
      await this.sendNotification(result);
    }
  }

  private async sendNotification(result: SkillResult) {
    // 集成 Slack/钉钉/邮件通知
  }
}

// src/core/base-skill.ts
export abstract class BaseSkill {
  abstract id: string;
  abstract name: string;
  abstract description: string;
  
  protected config: any;
  protected logger: Logger;

  constructor() {
    this.logger = new Logger(this.name);
  }

  abstract execute(context: SkillContext): Promise<SkillResult>;
  
  protected async callAI(prompt: string, options?: any) {
    // 统一的 AI 调用接口
    const { OpenAI } = await import('openai');
    
    const openai = new OpenAI({
      apiKey: process.env.OPENAI_API_KEY,
    });

    const completion = await openai.chat.completions.create({
      model: options?.model || 'gpt-4',
      messages: [
        { role: 'system', content: this.getSystemPrompt() },
        { role: 'user', content: prompt }
      ],
      temperature: options?.temperature || 0.7,
      max_tokens: options?.maxTokens || 1000,
    });

    return completion.choices[0].message.content;
  }

  protected abstract getSystemPrompt(): string;
}
2. 代码审查技能实现

// src/skills/code-review.ts
import { BaseSkill } from '../core/base-skill';
import * as fs from 'fs';
import * as path from 'path';

export class CodeReviewSkill extends BaseSkill {
  id = 'code-review';
  name = '代码审查助手';
  description = '自动审查代码质量和最佳实践';

  async execute(context: SkillContext): Promise<SkillResult> {
    const { filePath, rules = [] } = context.options;
    
    const code = await this.readFile(filePath);
    const review = await this.reviewCode(code, rules);
    
    return {
      skillId: this.id,
      success: true,
      output: review,
      suggestions: review.suggestions,
      metadata: {
        file: filePath,
        issuesCount: review.issues.length
      }
    };
  }

  private async readFile(filePath: string): Promise<string> {
    return fs.promises.readFile(path.resolve(filePath), 'utf-8');
  }

  private async reviewCode(code: string, rules: string[]): Promise<any> {
    const prompt = `
请审查以下前端代码，关注：
1. 代码质量和可读性
2. 性能优化
3. 安全最佳实践
4. 符合 React/Vue 最佳实践
5. TypeScript 类型安全

代码：
\`\`\`typescript
${code}
\`\`\`

请按以下格式返回：
1. 总体评价
2. 具体问题（按严重程度排序）
3. 改进建议
    `;

    const aiResponse = await this.callAI(prompt);
    return this.parseReviewResponse(aiResponse);
  }

  protected getSystemPrompt(): string {
    return `你是一个经验丰富的前端开发专家，擅长代码审查和最佳实践。请以专业、建设性的方式提供反馈。`;
  }

  private parseReviewResponse(response: string) {
    // 解析 AI 返回的结构化数据
    // 这里可以添加更复杂的解析逻辑
    return {
      summary: '',
      issues: [],
      suggestions: []
    };
  }
}
3. 组件生成技能
typescript
// src/skills/component-gen.ts
export class ComponentGenSkill extends BaseSkill {
  id = 'component-gen';
  name = '组件生成器';
  description = '根据描述生成 React/Vue 组件代码';

  async execute(context: SkillContext): Promise<SkillResult> {
    const {
      description,
      framework = 'react',
      type = 'tsx',
      style = 'css'
    } = context.options;

    const prompt = this.buildComponentPrompt(description, framework, type, style);
    const code = await this.callAI(prompt);
    
    const filePath = await this.saveComponent(code, context.options.name);
    
    return {
      skillId: this.id,
      success: true,
      output: code,
      metadata: {
        filePath,
        framework,
        type
      }
    };
  }

  private buildComponentPrompt(
    description: string,
    framework: string,
    type: string,
    style: string
  ): string {
    return `
作为前端专家，请生成一个 ${framework.toUpperCase()} 组件。

要求：
1. 组件描述：${description}
2. 使用 ${type === 'tsx' ? 'TypeScript' : 'JavaScript'}
3. 样式使用 ${style}
4. 包含完整的类型定义（如果是TS）
5. 添加必要的注释
6. 遵循 ${framework} 最佳实践
7. 组件应该是可复用的

请只返回代码，不要有额外解释。
    `;
  }

  private async saveComponent(code: string, componentName: string): Promise<string> {
    const fileName = `${componentName}.${this.getFileExtension()}`;
    const filePath = path.join(process.cwd(), 'components', fileName);
    
    // 确保目录存在
    await fs.promises.mkdir(path.dirname(filePath), { recursive: true });
    await fs.promises.writeFile(filePath, code, 'utf-8');
    
    return filePath;
  }

  protected getSystemPrompt(): string {
    return `你是前端组件开发专家，精通 React、Vue、TypeScript 和现代 CSS。请生成高质量、可维护的组件代码。`;
  }
}

export class BugFixSkill extends BaseSkill {

      output: fix,
      metadata: {
        bugType: analysis.type,
        severity: analysis.severity,
        suggestedFix: fix.code
      }
    };

分析以下错误并提供修复方案：

错误信息：
${errorLog}

相关代码：
\`\`\`
${code}
\`\`\`

环境：${env}

请分析：
1. 错误类型和原因
2. 影响范围
3. 修复建议
    `;

    return this.callAI(prompt);
  }
}
